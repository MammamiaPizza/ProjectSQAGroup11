  private Node findObjectLiteral(Node node) {
    if (node == null) {
      return null;
    }
    if (node.isObjectLit()) {
      return node;
    }
    for (Node child = node.getFirstChild(); child != null; child = child.getNext()) {
      Node result = findObjectLiteral(child);
      if (result != null) {
        return result;
      }
    }
    return null;
  }
