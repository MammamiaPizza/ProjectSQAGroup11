    public void testAtan2SignedZeroQuadrants() {

        DerivativeStructure negativeZero = new DerivativeStructure(1, 1, 0, -0.0);

        Assert.assertEquals(Math.PI, positiveZero.atan2(negativeZero).getValue(), 0.0);
        Assert.assertEquals(-Math.PI, negativeZero.atan2(negativeZero).getValue(), 0.0);

        double negativeZeroResult = negativeZero.atan2(positiveZero).getValue();
        Assert.assertEquals(Double.doubleToLongBits(-0.0),
                            Double.doubleToLongBits(negativeZeroResult));
    }
