    private Option createOption(final int maximumArguments) {

                .withName("value")
                .withMinimum(1)
                .withMaximum(maximumArguments)
                .create();

        return new DefaultOptionBuilder()
                .withShortName("v")
                .withArgument(argument)
                .create();
    }
