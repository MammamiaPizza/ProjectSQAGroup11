    @Test(expected = MathArithmeticException.class)
    public void multiplyIntMustRejectNumeratorOverflow() {

    }

    @Test(expected = MathArithmeticException.class)

    }

    @Test(expected = MathArithmeticException.class)
    public void fractionAdditionMustRejectOverflowingResult() {

    @Test(expected = MathArithmeticException.class)
