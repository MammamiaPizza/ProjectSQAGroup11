private int binarySearch(int key) {
 int low = 0;
 int high = size - 1;

 while (low <= high) {
 int mid = (low + high) >> 1;
 int midVal = values[mid];

 if (midVal < key) {
 low = mid + 1;
 } else if (midVal > key) {
 high = mid - 1;
 } else {
 return mid; // key found
 }
 }
 return -(low + 1); // key not found.
