     private double[][] multiply(double[][] a, double[][] b) {
         int m = a.length;
         int n = a[0].length;
         int p = b[0].length;
         double[][] result = new double[m][p];
         for (int i = 0; i < m; i++) {
             for (int k = 0; k < n; k++) {
                 double aik = a[i][k];
                 for (int j = 0; j < p; j++) {
                     result[i][j] += aik * b[k][j];
                 }
             }
         }
         return result;
     }

     private double[][] subtract(double[][] a, double[][] b) {
         int m = a.length;
         int n = a[0].length;
         double[][] result = new double[m][n];
         for (int i = 0; i < m; i++) {
             for (int j = 0; j < n; j++) {
                 result[i][j] = a[i][j] - b[i][j];
             }
         }
         return result;
     }
