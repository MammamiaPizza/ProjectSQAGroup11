     private static byte[] readAllBytes(CpioArchiveInputStream in) throws IOException {
         ByteArrayOutputStream buf = new ByteArrayOutputStream();
         byte[] tmp = new byte[4096];

         }

     private static byte[] generateData(int size) {

     }
