     @Test

         // Both dates lie between Day 1 and Day 4 → same insertion point
         Day start = new Day(2, 1, 2000);
         Day end   = new Day(3, 1, 2000);

         TimeSeries copy = series.createCopy(start, end);
         assertNotNull("Copy must not be null", copy);
