     private static MultivariateRealFunction sphere(final int dim) {

             public double value(double[] x) {
                 double s = 0.0;
                 for (double v : x) s += v * v;

             }

             public double value(double[] x) {
                 double a = 1.0 - x[0];

             }
         };
